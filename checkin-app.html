<script>

(function(){

  // ====== SETUP: paste your deployed Apps Script web app URL below ======

  const API_URL = 'PASTE_YOUR_APPS_SCRIPT_URL_HERE';

  // ========================================================================


  if(API_URL.indexOf('PASTE_YOUR') === 0){

    document.getElementById('setupBanner').style.display = 'block';

  }


  const GRADE_LIST = ['Nursery','Pre-K1','Pre-K2','Kinder','G1','G2','G3','G4','G5'];

  const PYP_GRADES = new Set(['Kinder','G1','G2','G3','G4','G5']);

  function programForGrade(grade){ return PYP_GRADES.has(grade) ? 'PYP' : 'EYP'; }


  let roster = [];

  let queues = {EYP:[], PYP:[]};


  function loadData(){

    fetch(API_URL).then(r=>r.json()).then(data=>{

      roster = data.roster || [];

      queues = data.queues || {EYP:[], PYP:[]};

      renderRosterTable();

      renderQueues();

    }).catch(err=>console.error('load failed', err));

  }


  function postAction(payload){

    // text/plain avoids a CORS preflight; the server parses it as JSON regardless

    return fetch(API_URL, {

      method:'POST',

      headers:{'Content-Type':'text/plain;charset=utf-8'},

      body: JSON.stringify(payload)

    }).then(r=>r.json());

  }


  function renderQueues(){

    renderBoard('boardEyp', queues.EYP); renderBoard('boardPyp', queues.PYP);

    renderMini('miniEyp', queues.EYP); renderMini('miniPyp', queues.PYP);

  }


  function renderBoard(id, list){

    const el = document.getElementById(id); el.innerHTML = '';

    for(let i=0;i<10;i++){

      const item = list[i]; const div = document.createElement('div');

      if(item){

        div.className = 'slot' + (i===0 ? ' new' : '');

        div.innerHTML = `<div class="num">${i+1}</div>

          <div><div class="name">${escapeHtml(item.name)}</div><div class="code">${escapeHtml(item.grade||'')} · ${escapeHtml(item.code)}</div></div>

          <div class="time">${escapeHtml(item.time)}</div>`;

      } else {

        div.className = 'slot empty';

        div.innerHTML = `<div class="num">${i+1}</div><div class="name">—</div><div></div>`;

      }

      el.appendChild(div);

    }

  }


  function renderMini(id, list){

    const el = document.getElementById(id); el.innerHTML = '';

    if(!list.length){ el.innerHTML = '<li class="empty-row">No check-ins yet.</li>'; return; }

    list.slice(0,5).forEach(item=>{

      const li = document.createElement('li');

      li.innerHTML = `<span>${escapeHtml(item.name)} <span class="code">${escapeHtml(item.grade||'')}</span></span><span class="code">${escapeHtml(item.time)}</span>`;

      el.appendChild(li);

    });

  }


  function escapeHtml(s){ return (s||'').toString().replace(/[&<>"']/g, c => ({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[c])); }


  function handleScan(rawCode){

    const code = (rawCode||'').trim(); if(!code) return;

    const fb = document.getElementById('feedback');

    postAction({action:'scan', code:code}).then(result=>{

      if(result.ok){

        fb.className = 'feedback ok' + (result.program==='PYP' ? ' pyp' : '');

        fb.innerHTML = `<div><div class="fb-name">${escapeHtml(result.name)}</div><div class="fb-sub">${escapeHtml(result.grade||'')} · Checked in · ${escapeHtml(result.time)}</div></div>

          <span class="fb-badge ${result.program==='PYP'?'pyp':''}">${result.program}</span>`;

        loadData();

      } else {

        fb.className = 'feedback bad';

        fb.innerHTML = `<div><div class="fb-name">Code not found</div><div class="fb-sub">"${escapeHtml(code)}" isn't in the roster yet — add it under Manage roster.</div></div>`;

      }

    }).catch(err=>{

      fb.className = 'feedback bad';

      fb.innerHTML = `<div><div class="fb-name">Couldn't reach the sheet</div><div class="fb-sub">${escapeHtml(String(err))}</div></div>`;

    });

  }


  function renderRosterTable(){

    const body = document.getElementById('rosterBody'); body.innerHTML = '';

    roster.forEach((r, idx)=>{

      const tr = document.createElement('tr');

      const gradeOptions = GRADE_LIST.map(g=>`<option value="${g}" ${r.grade===g?'selected':''}>${g}</option>`).join('');

      const program = programForGrade(r.grade);

      tr.innerHTML = `

        <td><input type="text" data-idx="${idx}" data-field="code" value="${escapeHtml(r.code)}"></td>

        <td><input type="text" data-idx="${idx}" data-field="name" value="${escapeHtml(r.name)}"></td>

        <td><select data-idx="${idx}" data-field="grade">${gradeOptions}</select></td>

        <td><span class="pill ${program==='PYP'?'pyp':'eyp'}">${program}</span></td>

        <td><button class="row-del" data-idx="${idx}">Remove</button></td>`;

      body.appendChild(tr);

    });

  }


  document.getElementById('rosterBody').addEventListener('input', e=>{

    const idx = e.target.getAttribute('data-idx'), field = e.target.getAttribute('data-field');

    if(idx===null) return; roster[idx][field] = e.target.value;

  });

  document.getElementById('rosterBody').addEventListener('change', e=>{

    const idx = e.target.getAttribute('data-idx'), field = e.target.getAttribute('data-field');

    if(idx===null) return; roster[idx][field] = e.target.value;

    if(field === 'grade') renderRosterTable();

  });

  document.getElementById('rosterBody').addEventListener('click', e=>{

    if(e.target.classList.contains('row-del')){ roster.splice(e.target.getAttribute('data-idx'),1); renderRosterTable(); }

  });

  document.getElementById('addRowBtn').addEventListener('click', ()=>{ roster.push({code:'', name:'', grade:'Nursery'}); renderRosterTable(); });

  document.getElementById('saveRosterBtn').addEventListener('click', ()=>{

    const note = document.getElementById('rosterNote'); note.textContent = 'Saving…';

    postAction({action:'saveRoster', roster:roster}).then(()=>{

      note.textContent = 'Saved.'; setTimeout(()=>{ note.textContent=''; }, 1800);

    }).catch(err=>{ note.textContent = 'Save failed — ' + err; });

  });

  document.getElementById('bulkAddBtn').addEventListener('click', ()=>{

    const lines = document.getElementById('bulkText').value.split('\n').map(l=>l.trim()).filter(Boolean);

    lines.forEach(line=>{

      const parts = line.split(',').map(p=>p.trim());

      if(parts.length>=2){

        const match = GRADE_LIST.find(g => g.toLowerCase() === (parts[2]||'').toLowerCase());

        roster.push({code:parts[0], name:parts[1], grade: match || 'Nursery'});

      }

    });

    document.getElementById('bulkText').value = ''; renderRosterTable();

  });


  const scanInput = document.getElementById('scanInput');

  scanInput.addEventListener('keydown', e=>{ if(e.key === 'Enter'){ handleScan(scanInput.value); scanInput.value = ''; } });

  document.addEventListener('click', (e)=>{

    const activeView = document.querySelector('.view.active');

    if(activeView && activeView.id === 'view-scan' && !e.target.closest('input, select, textarea, button')){ scanInput.focus(); }

  });


  document.querySelectorAll('.tab').forEach(tab=>{

    tab.addEventListener('click', ()=>{

      document.querySelectorAll('.tab').forEach(t=>t.classList.remove('active'));

      document.querySelectorAll('.view').forEach(v=>v.classList.remove('active'));

      tab.classList.add('active');

      document.getElementById('view-' + tab.dataset.view).classList.add('active');

      if(tab.dataset.view === 'scan'){ scanInput.focus(); }

    });

  });


  let qrScanner = null, nativeStream = null, nativeLoopId = null, nativeDetector = null;

  let lastNativeValue = '', lastNativeAt = 0;


  function stopCamera(){

    const video = document.getElementById('camVideo'), btn = document.getElementById('camBtn');

    if(qrScanner){ qrScanner.stop(); qrScanner = null; }

    if(nativeLoopId){ cancelAnimationFrame(nativeLoopId); nativeLoopId = null; }

    if(nativeStream){ nativeStream.getTracks().forEach(t=>t.stop()); nativeStream = null; }

    video.style.display = 'none'; btn.textContent = 'Try camera scan';

  }


  async function startNativeDetector(video, note){

    nativeDetector = new BarcodeDetector({formats:['code_128','code_39','ean_13','ean_8','upc_a','upc_e','itf','qr_code','pdf417','data_matrix']});

    nativeStream = await navigator.mediaDevices.getUserMedia({video:{facingMode:'environment'}});

    video.srcObject = nativeStream; video.style.display = 'block'; await video.play();

    note.textContent = 'Point a barcode or QR code at the camera.';

    const loop = async ()=>{

      if(!nativeStream) return;

      try{

        const codes = await nativeDetector.detect(video);

        if(codes && codes.length){

          const value = codes[0].rawValue, now = Date.now();

          if(value && (value !== lastNativeValue || now - lastNativeAt > 1500)){

            lastNativeValue = value; lastNativeAt = now; handleScan(value);

          }

        }

      }catch(e){}

      nativeLoopId = requestAnimationFrame(loop);

    };

    loop();

  }


  async function startQrFallback(video, note){

    if(typeof QrScanner === 'undefined') throw new Error('QrScanner library unavailable');

    QrScanner.WORKER_PATH = 'https://cdnjs.cloudflare.com/ajax/libs/qr-scanner/1.4.2/qr-scanner-worker.min.js';

    video.style.display = 'block';

    qrScanner = new QrScanner(video, result=>{ handleScan(typeof result === 'string' ? result : result.data); }, { returnDetailedScanResult:true });

    await qrScanner.start();

    note.textContent = 'Point a QR code at the camera. This browser doesn\'t support native barcode scanning, so only QR codes work here.';

  }


  document.getElementById('camBtn').addEventListener('click', async ()=>{

    const video = document.getElementById('camVideo'), note = document.getElementById('camNote'), btn = document.getElementById('camBtn');

    if(qrScanner || nativeStream){ stopCamera(); return; }

    try{

      if(typeof BarcodeDetector !== 'undefined'){ await startNativeDetector(video, note); }

      else { await startQrFallback(video, note); }

      btn.textContent = 'Stop camera scan';

    }catch(err){

      note.textContent = 'Camera permission was blocked or unavailable. Use a USB/Bluetooth scanner or manual entry instead.';

      video.style.display = 'none';

    }

  });


  setInterval(loadData, 2500);

  loadData();

})();
